\begin{frame}{Motivation}
  \begin{itemize}
    \item State the economic setting and why it matters.
    \item Identify the unresolved empirical or theoretical problem.
    \item End with the precise research question.
  \end{itemize}
\end{frame}

\begin{frame}{This Paper}
  \begin{itemize}
    \item State the research design in one sentence.
    \item Summarize the main verified finding without unsupported numbers.
    \item Explain what the finding changes about the question.
  \end{itemize}
\end{frame}

\begin{frame}{Contributions}
  \begin{enumerate}
    \item Position the first contribution relative to verified literature.
    \item State the empirical or measurement contribution.
    \item State the broader economic implication.
  \end{enumerate}
\end{frame}
