\section{Background}
\subsection{Setting}

\begin{frame}{Institutional Setting}
  \begin{itemize}
    \item Introduce only the institutional facts needed for identification.
    \item Use a map, timeline, or process diagram when supplied.
    \item Separate descriptive context from causal assumptions.
  \end{itemize}
\end{frame}

\subsection{Variation}

\begin{frame}{Source of Variation}
  \begin{itemize}
    \item Describe what varies, across which units, and over what period.
    \item Explain why that variation is economically meaningful.
  \end{itemize}
\end{frame}
