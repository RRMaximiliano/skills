\section{Mechanisms}
\subsection{Evidence}

\begin{frame}{Mechanism Evidence}
  \begin{itemize}
    \item State the mechanism prediction.
    \item Present the strongest supplied test.
    \item Explain what the evidence can and cannot establish.
  \end{itemize}
\end{frame}

\subsection{Alternatives}

\begin{frame}{Alternative Explanations}
  \begin{itemize}
    \item Name the most credible alternative channel.
    \item Show the supplied evidence that distinguishes it from the main account.
  \end{itemize}
\end{frame}
